# Generates compile_commands.json for clangd and clang-tidy. Nothing is built.

cmake_minimum_required(VERSION 3.12.0)

# CMake 4 sets no macOS SDK, so Homebrew clang tools can't find the standard library.
if(APPLE)
  execute_process(
    COMMAND xcrun --sdk macosx --show-sdk-path
    OUTPUT_VARIABLE MACOS_SDK_PATH
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
  if(MACOS_SDK_PATH)
    add_compile_options(-isysroot "${MACOS_SDK_PATH}")
  endif()
endif()

project(rnaudioworklets-clangd LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 20)

set(COMMON_CPP_DIR ${CMAKE_SOURCE_DIR}/..)
set(WORKLETS_PACKAGE_DIR ${CMAKE_SOURCE_DIR}/../../..)
set(REPO_ROOT_DIR ${CMAKE_SOURCE_DIR}/../../../../../)
set(REACT_NATIVE_DIR ${WORKLETS_PACKAGE_DIR}/node_modules/react-native)
set(WORKLETS_DIR ${WORKLETS_PACKAGE_DIR}/node_modules/react-native-worklets/Common/cpp)
set(AUDIO_API_COMMON_CPP_DIR ${REPO_ROOT_DIR}/packages/react-native-audio-api/common/cpp)

set(CODEGEN_JNI_DIR "${WORKLETS_PACKAGE_DIR}/android/build/generated/source/codegen/jni")
set(CODEGEN_IOS_DIR "${REPO_ROOT_DIR}/apps/fabric-example/ios/build/generated/ios/ReactCodegen")
# rnaudioworkletsJSI.h, generated by generate-and-copy.sh.
set(CODEGEN_LOCAL_DIR "${CMAKE_BINARY_DIR}/codegen/build/generated/ios/ReactCodegen")

file(GLOB_RECURSE AUDIO_WORKLETS_SOURCES CONFIGURE_DEPENDS
  "${COMMON_CPP_DIR}/audioworklets/*.cpp")

add_library(audioworklets_compile_db STATIC ${AUDIO_WORKLETS_SOURCES})

# REACT_NATIVE_PRODUCTION stops react_native_assert.h from including glog (Pods only).
target_compile_definitions(audioworklets_compile_db PRIVATE RCT_NEW_ARCH_ENABLED REACT_NATIVE_PRODUCTION)

set(FABRIC_EXAMPLE_IOS_DIR ${REPO_ROOT_DIR}/apps/fabric-example/ios)
set(PODS_HEADERS_DIR "${FABRIC_EXAMPLE_IOS_DIR}/Pods/Headers/Public")

target_include_directories(
  audioworklets_compile_db
  PRIVATE
  "${COMMON_CPP_DIR}"
  "${AUDIO_API_COMMON_CPP_DIR}"
  "${WORKLETS_DIR}"
  "${REACT_NATIVE_DIR}/ReactCommon"
  "${REACT_NATIVE_DIR}/ReactCommon/jsi"
  "${REACT_NATIVE_DIR}/ReactCommon/callinvoker"
  "${REACT_NATIVE_DIR}/ReactCommon/react/nativemodule/core"
  "${REACT_NATIVE_DIR}/ReactCommon/react/bridging"
  "${CODEGEN_JNI_DIR}/react/renderer/components/rnaudioworklets"
  "${CODEGEN_LOCAL_DIR}"
  "${CODEGEN_IOS_DIR}"
  "${PODS_HEADERS_DIR}/ReactNativeDependencies"
  "${PODS_HEADERS_DIR}/React-Core"
)
